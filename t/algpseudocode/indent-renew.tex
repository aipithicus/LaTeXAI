\caption{Renewable indentation}\label{alg:indent}
\begin{algorithmic}[1]
\Procedure{Extract}{$D$}
  \State Outer: \IndentValue.\label{line:outer}
  \algrenewcommand\algorithmicindent{\stepvalue}
  \ForEach{$x\in D$}
    \State Inner: \IndentValue.
    \setlength{\stepwidth}{7pt}
  \EndFor
  \State After inner: \IndentValue.
\EndProcedure
\State After outer: \IndentValue.
\If{$D\ne 0$}
  \State New value: \IndentValue.
\EndIf
\State After new value: \IndentValue.
\end{algorithmic}
\end{algorithm}
