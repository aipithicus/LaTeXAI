% algorithmicx 1.2:681-700 saves line/frequency/remainder and open indents.
% Line 3 leaves a nonzero remainder at frequency 2. Both open blocks must
% survive the environment group, and later closes use their saved widths.
\begin{algorithm}
\begin{algorithmic}[2]
\algrenewcommand\algorithmicindent{1em}
\Procedure{Split}{$x$}
\algrenewcommand\algorithmicindent{2em}
\If{$x>0$}
\State $x\gets x+1$\label{split:before}
\algstore{split}
\end{algorithmic}
\begin{algorithmic}[1]
\algrenewcommand\algorithmicindent{4em}
\algrestore{split}
\State $x\gets x+2$\label{split:after}
\EndIf
\State \Return $x$
\EndProcedure
\end{algorithmic}
\end{algorithm}
